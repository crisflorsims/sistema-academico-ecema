export const getEstadisticasAprobacion = async (req: Request, res: Response): Promise<void> => {
  try {
    const inscripciones = await prisma.inscripcionMateria.findMany({
      include: {
        calificaciones: true,
        materia: {
          include: {
            diplomado: true,
            area: true
          }
        }
      }
    });

    const statsDiplomados: Record<string, { aprobados: number, reprobados: number }> = {};
    const statsAreas: Record<string, { aprobados: number, reprobados: number }> = {};

    for (const insc of inscripciones) {
      if (insc.calificaciones.length === 0) continue; // Ignorar si no tiene notas

      const totalNota = insc.calificaciones.reduce((sum, cal) => sum + cal.nota, 0);
      const diplomadoNombre = insc.materia.diplomado.nombre;
      const areaNombre = insc.materia.area?.nombre || 'Sin Área';

      let notaMinima = 71;
      const dNombreUpper = diplomadoNombre.toUpperCase();
      if (dNombreUpper.includes('DCEMA')) {
        notaMinima = 80;
      } else if (dNombreUpper.includes('DCEA')) {
        notaMinima = 75;
      }

      const esAprobado = totalNota >= notaMinima;

      if (!statsDiplomados[diplomadoNombre]) {
        statsDiplomados[diplomadoNombre] = { aprobados: 0, reprobados: 0 };
      }
      if (!statsAreas[areaNombre]) {
        statsAreas[areaNombre] = { aprobados: 0, reprobados: 0 };
      }

      if (esAprobado) {
        statsDiplomados[diplomadoNombre].aprobados++;
        statsAreas[areaNombre].aprobados++;
      } else {
        statsDiplomados[diplomadoNombre].reprobados++;
        statsAreas[areaNombre].reprobados++;
      }
    }

    // Convertir a arreglos para Recharts
    const dataDiplomados = Object.keys(statsDiplomados).map(key => ({
      name: key,
      Aprobados: statsDiplomados[key].aprobados,
      Reprobados: statsDiplomados[key].reprobados
    }));

    const dataAreas = Object.keys(statsAreas).map(key => ({
      name: key,
      Aprobados: statsAreas[key].aprobados,
      Reprobados: statsAreas[key].reprobados
    }));

    res.json({ diplomados: dataDiplomados, areas: dataAreas });
  } catch (error) {
    console.error("Error al obtener estadísticas de aprobación:", error);
    res.status(500).json({ error: 'Error interno al generar estadísticas de aprobación' });
  }
};
